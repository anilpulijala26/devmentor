    type UserRole = "admin" | "instructor" | "student";

      readonly id: number;
      name: string;
      email: string;
      role: UserRole;
      phoneNumber?: string; // Optional property

      if (user.phoneNumber) {

      return `${user.name} has no phone. Contact email is: ${user.email}`;
    };

    // 4. Union & Narrowing Logic
    const formatIdentifier = (id: string | number): string => {
      if (typeof id === "string") {
        return `UUID_${id.toUpperCase()}`;
      }
      return `ID_#${id.toFixed(0)}`;
    };

## Basic Types

```typescript
// Primitives
const name: string = "Alice";
const age: number = 25;
const active: boolean = true;

// Arrays
const numbers: number[] = [1, 2, 3];
const strings: Array<string> = ["a", "b"];

// Union types
const id: string | number = 123;

// Any (avoid when possible)
let data: any = "anything";
```

## Interfaces and Types

```typescript
// Interface
interface User {
  name: string;
  age: number;
  email?: string;  // Optional
}

// Type alias
type UserRole = "admin" | "user" | "guest";

// Using them
const user: User = {
  name: "Alice",
  age: 25
};

const role: UserRole = "admin";
```

## Functions

```typescript
// Function with types
const add = (a: number, b: number): number => {
  return a + b;
};

// Function with optional parameters
const greet = (name: string, greeting: string = "Hello"): string => {
  return `${greeting}, ${name}`;
};
```

<SeniorNote>
In production React apps, you'll use interfaces extensively for component props. TypeScript prevents passing wrong types and makes your code self-documenting.
</SeniorNote>

<Pitfall>
Overusing `any` defeats the purpose of TypeScript. If you're unsure, use `unknown` and narrow it explicitly.
</Pitfall>

## Key Takeaways

- Use types for function parameters and return values
- Define interfaces for complex objects
- Use union types for multiple possibilities
- TypeScript catches type errors at compile time
